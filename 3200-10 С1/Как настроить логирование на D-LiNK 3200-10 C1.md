// Как настроить логирование на D-Link 3200-10 C1 ?  

config log_save_timing on_demand  
enable syslog  
config system_severity log information  
create syslog host 1 ipaddress 10.245.2.210 severity informational facility local0 udp_port 514 state enable  
enable command logging  

save
